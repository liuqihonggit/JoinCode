   ```csharp
   {
       public TimeSpan Timeout { get; set; } = TimeSpan.FromSeconds(30);
       public int MaxRetries { get; set; } = 3;
       public TimeSpan InitialBackoff { get; set; } = TimeSpan.FromSeconds(1);
       public TimeSpan MaxBackoff { get; set; } = TimeSpan.FromSeconds(10);
   }
   ```

2. 实现指数退避重试
   ```csharp
   private async Task<ProcessResult> ExecuteWithRetryAsync(
       ProcessOptions options, CancellationToken ct)
   {
       var retryCount = 0;
       while (retryCount <= _maxRetries)
       {
           try
           {
               return await _processService.ExecuteAsync(options, ct);
           }
           catch (OperationCanceledException) when (!ct.IsCancellationRequested)
           {
               if (retryCount >= _maxRetries) throw;
               var delay = TimeSpan.FromSeconds(
                   Math.Min(_initialBackoff.TotalSeconds * Math.Pow(2, retryCount),
                            _maxBackoff.TotalSeconds));
               await Task.Delay(delay, ct);
               retryCount++;
           }
       }
   }
   ```

2. 实现改写器注册表
   ```csharp
   public class CommandRewriterRegistry
   {
       private readonly List<ICommandRewriter> _rewriters = new();
       
       public void Register(ICommandRewriter rewriter) => _rewriters.Add(rewriter);
       
       public string Rewrite(string command, Dictionary<string, object> context)
       {
           foreach (var rewriter in _rewriters.OrderByDescending(r => r.Priority))
           {
               if (rewriter.CanRewrite(command))
               {
                   return rewriter.Rewrite(command, context);
               }
           }
           return command;
       }
   }
   ```

   ```csharp
   public class ToolErrorTracker
   {
       private readonly Dictionary<string, int> _errorCounts = new();
       private readonly Dictionary<string, DateTime> _lastErrorTime = new();
       private readonly int _threshold;
       
       public ToolErrorTracker(int threshold = 3)
       {
           _threshold = threshold;
       }
       
       public void RecordError(string toolName)
       {
           _errorCounts[toolName] = _errorCounts.GetValueOrDefault(toolName) + 1;
           _lastErrorTime[toolName] = DateTime.UtcNow;
       }
       
       public void RecordSuccess(string toolName)
       {
           _errorCounts.Remove(toolName);
       }
       
       public bool ShouldAutoFix(string toolName)
       {
           return _errorCounts.GetValueOrDefault(toolName) >= _threshold;
       }
       
       public int GetErrorCount(string toolName)
       {
           return _errorCounts.GetValueOrDefault(toolName);
       }
   }
   ```

2. 集成到 `ToolHealthScoringMiddleware`

1. 定义修正 hook 注册表
   ```csharp
   public class ToolFixHookRegistry
   {
       private readonly List<IToolFixHook> _hooks = new();
       private readonly ToolErrorTracker _tracker;
       
       public ToolFixHookRegistry(ToolErrorTracker tracker)
       {
           _tracker = tracker;
       }
       
       public void Register(IToolFixHook hook) => _hooks.Add(hook);
       
       public async Task<bool> TryFixAsync(
           string toolName, Exception error, CancellationToken ct)
       {
           if (!_tracker.ShouldAutoFix(toolName)) return false;
           
           foreach (var hook in _hooks.Where(h => h.CanFix(toolName, error)))
           {
               await hook.FixAsync(toolName, ct);
               return true;
           }
           return false;
       }
   }
   ```

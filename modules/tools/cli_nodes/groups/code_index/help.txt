单个工具的参数说明:

  code_index_search — C# AST symbol search ONLY. Searches indexed C# code symbols (classes, methods, properties, etc.) from parsed .cs files. Do NOT use for config files, docs, JSON, YAML, or non-C# content - use grep/glob instead.
    --query                      string    必填   Search query (supports FTS5 full-text search syntax)

  code_index_search_comprehensive — C# AST comprehensive search ONLY. Fuzzy match C# symbols then AST-extract references + callers/callees. Do NOT use for config, docs, JSON, YAML, or non-C# content - use grep/glob instead.
    --pattern                    string    必填   Regex pattern to fuzzy match symbol name/FQN (rg-style, e.g. 'User*' or 'Get.*Name')
    --max_token_budget           integer   必填   Max token budget for result (approx 4 chars/token, truncated if exceeded, default 2000)  默认=2000
    --include_ast                boolean   必填   Include AST expansion (references + callers/callees). Set false to only return symbol matches, saving tokens for non-code queries. Default true.  默认=true

  code_index_find_definition — Find the definition location of a C# symbol in AST index
    --symbol_name                string    必填   Symbol name

  code_index_find_references — Find all references to a C# symbol in AST index
    --symbol_name                string    必填   Symbol name
    --limit                      integer   必填   Maximum references to return (default 5, pass larger to see more)  默认=5

  code_index_get_callers — Find all callers of a specified C# symbol in AST index
    --symbol_name                string    必填   Symbol name
    --limit                      integer   必填   Maximum callers to return (default 5, pass larger to see more)  默认=5

  code_index_get_callees — Find all callees invoked by a specified C# symbol in AST index
    --symbol_name                string    必填   Symbol name
    --limit                      integer   必填   Maximum callees to return (default 5, pass larger to see more)  默认=5

  code_index_get_call_chain — Find the call chain between two symbols
    --from                       string    必填   Source symbol name
    --to                         string    必填   Target symbol name

  code_index_get_impact_scope — Analyze the impact scope of modifying a symbol
    --symbol_name                string    必填   Symbol name

  code_index_get_inheritors — Find types that inherit or implement a specified symbol
    --symbol_name                string    必填   Symbol name

  code_index_get_dependencies — Find dependencies of a specified symbol
    --symbol_name                string    必填   Symbol name

  code_index_get_affected_files — Analyze files affected by modifying a specified file
    --file_path                  string    必填   File path

  code_index_rebuild — Rebuild the code index
    --workspace_root             string    必填   Workspace root directory path
    --force                      boolean   必填   Force full rebuild, skip hash check  默认=false

  code_index_update_file — Incrementally update index for a single file. Much faster than full rebuild when only a few files changed.
    --file_path                  string    必填   File path to update (absolute or relative to workspace root)

  code_index_stats — Get code index statistics
    (无参数)

  code_index_explore — Progressively explore C# code: symbol index -> call relationships -> source code. AST index only.
    --query                      string    必填   Search query (symbol name or keyword)
    --level                      string    必填   Disclosure level: index=symbol index only, relationships=with call graph, source=with source code  默认=index

  code_index_search_semantic — Semantic search code blocks via vector embeddings. Find similar code by meaning, not exact text match. Returns metadata (file path, line range, symbol name, similarity score). Set include_source_text=true to get matched block source code. Set include_parent_document=true to get parent class/file source. Set include_graph=true to get caller/callee relations.
    --query                      string    必填   Natural language query or code snippet (e.g. 'find authentication logic', 'rate limiting implementation')
    --top_k                      integer   可选   Maximum number of results to return  默认=20
    --include_source_text        boolean   可选   Include matched block source text (function code) in results  默认=false
    --include_parent_document    boolean   可选   Include parent document (class/file) source text in results for full context  默认=false
    --include_graph              boolean   可选   Include knowledge graph triples (caller,calls,callee). GraphRAG: vector recall + graph triples correct AI cognition  默认=true
    --file_type                  string    可选   Filter by file extension without dot, e.g. 'cs' for C# only, 'md' for Markdown only. Default null = all file types
    --namespace_filter           string    可选   Filter by namespace prefix, e.g. 'JoinCode.CodeIndex' for code in that namespace only. Default null = all namespaces
    --symbol_kind                string    可选   Filter by symbol kind, e.g. 'Method' for methods only, 'Class' for classes only. Default null = all kinds
    --persist_dir                string    可选   Persistence directory path to load index from. Default null = auto-discover git workspace root

  code_index_get_project_deps — Find project dependencies of a specified project
    --project_path               string    必填   Project file path (.csproj)

  code_index_get_project_dependents — Find projects that depend on a specified project
    --project_path               string    必填   Project file path (.csproj)

  code_index_get_affected_projects — Analyze projects affected by modifying a specified file
    --file_path                  string    必填   File path

  code_index_get_project_nu_gets — Find NuGet packages referenced by a specified project
    --project_path               string    必填   Project file path (.csproj)

  code_index_get_nu_get_projects — Find all projects referencing a specified NuGet package
    --package_name               string    必填   NuGet package name

  code_index_get_all_projects — List all projects in the workspace
    (无参数)

  search_document — Document search: semantic search over .md files only. Returns matching markdown chunks.
    --query                      string    必填   Search query (natural language)
    --top_k                      integer   可选   Maximum results  默认=10
    --include_source_text        boolean   必填   Include source text in results  默认=false
    --persist_dir                string    必填   Index directory (default: auto-discover git workspace)
